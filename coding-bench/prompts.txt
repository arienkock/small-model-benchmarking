// debounce.ts
export function debounce(fn: (...args: any[]) => void, waitMs: number) {
    let timer: ReturnType<typeof setTimeout> | null = null;
    return (...args: any[]) => {
        if (timer !== null) {
            return; // already scheduled
        }
        timer = setTimeout(() => fn(args), waitMs);
        timer = null;
    };
}

The intended behavior: debounce(fn, waitMs) returns a function that delays calling fn until
waitMs milliseconds have passed since the LAST call, passing through the latest arguments,
