```python
def first_unique(s: str):
    """
    Return the first character in `s` that appears exactly once.
    If no such character exists, return None.

    The algorithm runs in O(n) time and uses O(k) extra space,
    where k is the number of distinct characters in `s`.
    """
    # First pass: count occurrences of each character
    counts = {}
    for ch in s:
        counts[ch] = counts.get(ch, 0) + 1

    # Second pass: find the first character with a count of 1
    for ch in s:
        if counts[ch] == 1:
            return ch

    # No unique character found
    return None
