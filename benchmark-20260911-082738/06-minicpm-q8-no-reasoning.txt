```python
def first_unique(s):
    """
    Returns the first character in s that occurs exactly once.
    Returns None if no such character exists.
    Runs in O(n) time.
    """
    count = {}
    for char in s:
        count[char] = count.get(char, 0) + 1
    
    for char in s:
        if count[char] == 1:
            return char
    return None
```
