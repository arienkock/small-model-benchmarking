```
def first_unique(s: str) -> Optional[str]:
    # Count frequencies
    freq = {}
    for ch in s:
        freq[ch] = freq.get(ch, 0) + 1
    # Find first char with freq == 1
    for ch in s:
        if freq[ch] == 1:
            return ch
    return None
```
